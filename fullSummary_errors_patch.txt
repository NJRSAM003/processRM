_REAL_ERROR_RE = re.compile(
    r'\b(SEVERE|ERROR|FATAL|Traceback|Segmentation fault|core dumped|'
    r'srun: error|sbatch: error|killed|abort)\b',
    re.IGNORECASE,
)


def _classify_err_file(err_file):
    """Returns 'error' / 'warn' / None based on .err content."""
    try:
        size = os.path.getsize(err_file)
    except OSError:
        return None
    if size < 50:
        return None
    try:
        with open(err_file, 'r', errors='replace') as f:
            content = f.read(50000)
    except OSError:
        return None
    if _REAL_ERROR_RE.search(content):
        return 'error'
    if 'WARN' in content or 'Warning' in content or 'warning' in content:
        return 'warn'
    return 'warn'


def collect_errors(workdir):
    errors_dir = os.path.join(workdir, 'errors')
    logs_dir = os.path.join(workdir, 'logs')

    by_stage = defaultdict(list)
    warnings_by_stage = defaultdict(list)

    if os.path.isdir(errors_dir):
        for stage in os.listdir(errors_dir):
            stage_path = os.path.join(errors_dir, stage)
            if os.path.isdir(stage_path):
                for err_file in glob.glob(os.path.join(stage_path, '*')):
                    by_stage[stage].append(err_file)

    if os.path.isdir(logs_dir):
        for err_file in glob.glob(os.path.join(logs_dir, '*.err')):
            kind = _classify_err_file(err_file)
            if kind is None:
                continue
            base = os.path.basename(err_file).lower()
            if 'rmsy' in base:
                stage = 'rmsynth/rmclean'
            elif 'merge' in base:
                stage = 'merge'
            elif 'orchestrate' in base:
                stage = 'orchestrator'
            else:
                stage = 'unknown'
            if kind == 'error':
                by_stage[stage].append(err_file)
            else:
                warnings_by_stage[stage].append(err_file)

    return dict(by_stage), dict(warnings_by_stage)

# ----------------------------------------------------------------------
# processRM default configuration  (RM-Tools-sigma backend)
#
# These defaults give you CLEAN FDF cubes ready for CARTA, cleaned to a
# spatially-varying per-pixel sigma threshold:
#   1. Extract Stokes I/Q/U from the full IQUV input cube
#   2. make_noise_map         -> 2D per-pixel noise map (PyBDSF on Stokes I)
#   3. rmsynth3d -N           -> dirty FDF + RMSF + FDF_noise_th map
#   4. rmclean3d -c -5 -N     -> CLEAN FDFs using cutoff = 5 * noise(y,x)
#   5. Merge chunks           -> full-size FITS cubes
#   6. fits2idia              -> HDF5 cubes, openable in CARTA
#
# Leave the defaults if that workflow is what you want. Tweak individual
# parameters only if you have a reason -- every flag below corresponds to
# an rmsynth3d / rmclean3d / make_noise_map option (see comments).
# ----------------------------------------------------------------------

crop = ''                         # Optional spatial crop "[width, height]" in px; requires 'pointing' or has no effect. IGNORED if region_file is set.
pointing = ''                     # Optional crop centre "[x, y]" in px; requires 'crop' or has no effect. IGNORED if region_file is set.

[chunking]

# Full help:  singularity exec ~/processRM/container/rm-env.sif rmsynth3d --help
[rmsynth]

[rmclean]

ncores = 1                        # --ncores: multiprocessing per task (also bump slurm.cpus_per_task)
internal_chunk = ''               # --chunk: pixels per multiprocessing batch (only if ncores > 1)
